function addQueue(name,costM,costC,costD,time,type,id){ if(game.queue.length>=3){message("File de construction pleine !");return;} if(!can(costM,costC,costD)){message("Ressources insuffisantes.");return;} spend(costM,costC,costD); game.queue.push({name,timeLeft:time,total:time,type,id}); save();render(); } function building(name,level,m,c,d,time,id){ return `

${name} — Niveau ${level}

Coût : ${m} métal, ${c} cristal, ${d} deutérium

`; } function renderBuildings(){ let b=game.buildings; document.getElementById("buildings").innerHTML= building("Mine de métal",b.metal,200*b.metal,100*b.metal,0,30,"metal")+ building("Mine de cristal",b.crystal,150*b.crystal,200*b.crystal,0,45,"crystal")+ building("Centrale solaire",b.solar,180*b.solar,80*b.solar,0,30,"solar")+ building("Chantier spatial",b.shipyard,300,200,100,60,"shipyard")+ building("Laboratoire scientifique",b.lab,600,400,0,60,"lab")+ `

File de construction : ${game.queue.length}/3

`+ game.queue.map((q,i)=>`

${q.name} — ${q.timeLeft}s restantes

`).join(""); } const researches=[ {id:"energy",name:"Technologie énergétique",m:300,c:300,d:0,time:60}, {id:"propulsion",name:"Propulsion spatiale",m:500,c:400,d:100,time:90}, {id:"defense",name:"Technologie de défense",m:400,c:500,d:0,time:75} ]; function startResearch(r){ if(game.buildings.lab<1){message("Construisez d'abord un laboratoire.");return;} if(game.queue.some(q=>q.type==="research")){message("Une recherche est déjà en cours.");return;} if(game.tech[r.id]>=1){message("Cette recherche est déjà terminée.");return;} addQueue(r.name,r.m,r.c,r.d,r.time,"research",r.id); } function renderResearch(){ document.getElementById("research").innerHTML=researches.map(r=> `

${r.name}

Coût : ${r.m} métal, ${r.c} cristal, ${r.d} deutérium

Niveau : ${game.tech[r.id]}

` ).join(""); } function buildShip(id){ if(game.buildings.shipyard<1){message("Construisez d'abord un chantier spatial.");return;} if(id==="or1"){ addQueue("Vaisseau éclaireur OR-1",500,300,100,60,"ship","or1"); }else{ if(game.tech.propulsion<1){message("La propulsion spatiale est nécessaire.");return;} addQueue("Intercepteur OR-2",1000,700,300,90,"ship","or2"); } } function renderFleet(){ document.getElementById("fleet").innerHTML=`

🛰️ Éclaireur OR-1

En service : ${game.ships.or1}

Coût : 500 métal, 300 cristal, 100 deutérium

🚀 Intercepteur OR-2

En service : ${game.ships.or2}

Nécessite la propulsion spatiale.

`; } function explore(id){ const p=planets.find(x=>x.id===id); if(!p||game.discovered.includes(id)){return;} if(game.mission){message("Une mission est déjà en cours.");return;} if(game.ships.or1<1){message("Il faut au moins un OR-1 pour explorer.");return;} if(!can(p.cost,0,0)){message("Métal insuffisant pour cette mission.");return;} spend(p.cost,0,0); game.mission={id,timeLeft:p.time,total:p.time}; message("Mission lancée vers "+p.name+" !"); save();render(); } function renderGalaxy(){ document.getElementById("galaxy").innerHTML=planets.map(p=>{ const known=game.discovered.includes(p.id); return `

${known?"🪐":"🔒"} ${known?p.name:"Planète inconnue"}

Système : ${p.system} • Coordonnées : [${p.coords}]

${known?p.desc:"Données inconnues. Envoyez un éclaireur pour la découvrir."}

${known?"Planète découverte":`

Coût : ${p.cost} métal • Durée : ${p.time} secondes

`}
`; }).join(""); } function renderMission(){ const box=document.getElementById("missions"); if(!game.mission){box.innerHTML="Aucune mission en cours.";return;} const p=planets.find(x=>x.id===game.mission.id); box.innerHTML=`

🛰️ Exploration : ${p.name}

Temps restant : ${game.mission.timeLeft} secondes

`; } function render(){ document.getElementById("metal").textContent=fmt(game.metal); document.getElementById("crystal").textContent=fmt(game.crystal); document.getElementById("deut").textContent=fmt(game.deut); document.getElementById("energy").textContent=fmt(game.energy); renderBuildings();renderResearch();renderFleet();renderGalaxy();renderMission(); } function completeQueue(){ const now=[]; game.queue.forEach(q=>{ if(q.timeLeft<=0){ if(q.type==="building")game.buildings[q.id]++; if(q.type==="research")game.tech[q.id]++; if(q.type==="ship")game.ships[q.id]++; message(q.name+" terminé !"); }else now.push(q); }); game.queue=now; } function tick(){ game.metal+=game.buildings.metal*0.5; game.crystal+=game.buildings.crystal*0.25; game.deut+=game.buildings.solar*0.02; game.energy=14+game.buildings.solar*5; game.queue.forEach(q=>q.timeLeft--); completeQueue(); if(game.mission){ game.mission.timeLeft--; if(game.mission.timeLeft<=0){ const p=planets.find(x=>x.id===game.mission.id); if(p&&!game.discovered.includes(p.id)){ game.discovered.push(p.id); game.metal+=p.reward; message(p.name+" découverte ! Récompense : "+p.reward+" métal."); } game.mission=null; } } save();render(); } render(); setInterval(tick,1000);
